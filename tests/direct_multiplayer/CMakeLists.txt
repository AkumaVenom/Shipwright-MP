cmake_minimum_required(VERSION 3.20)
project(ShipwrightDirectTests LANGUAGES CXX)
find_package(Threads REQUIRED)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(DIRECT_SANITIZERS "Enable address/undefined-behavior sanitizers" OFF)
add_executable(direct_tests direct_tests.cpp ../../soh/soh/Network/Direct/Transport.cpp)
target_link_libraries(direct_tests PRIVATE Threads::Threads)
if(WIN32)
  target_link_libraries(direct_tests PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(direct_tests PRIVATE /W4)
else()
  target_compile_options(direct_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(DIRECT_SANITIZERS)
    target_compile_options(direct_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(direct_tests PRIVATE -fsanitize=address,undefined)
  endif()
endif()
enable_testing()
add_test(NAME direct_transport_and_progress COMMAND direct_tests)
set_tests_properties(direct_transport_and_progress PROPERTIES TIMEOUT 90)
